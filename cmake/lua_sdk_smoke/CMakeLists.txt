cmake_minimum_required(VERSION 3.21)
project(lonejson_lua_sdk_smoke C)

include("${LONEJSON_REPO_ROOT}/cmake/LonejsonDevelopmentRuntime.cmake")
include("${LONEJSON_REPO_ROOT}/cmake/LonejsonLinkWarnings.cmake")
set(LONEJSON_DEVELOPMENT_RUNTIME_EXTRA_DIRS "${LONEJSON_CORE_LIBDIR}")
add_executable(installed_lua "${LONEJSON_REPO_ROOT}/tests/test_lua_embedded_runner.c")
target_include_directories(installed_lua PRIVATE "${LONEJSON_C_PKT_SYSTEMS_ROOT}/include")
target_link_libraries(installed_lua PRIVATE "${LONEJSON_C_PKT_SYSTEMS_ROOT}/lib/liblua.so")
target_compile_options(installed_lua PRIVATE -std=gnu89 -Wall -Wextra -Werror)
lonejson_configure_link_warnings(installed_lua)
lonejson_configure_development_runtime(installed_lua)
